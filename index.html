// Function called when submitting the form async function registerUser(event) { event.preventDefault(); const name = document.getElementById('nameInput').value; const email = document.getElementById('emailInput').value; const type = document.getElementById('typeSelect').value; // Technical or Non-Technical // Send data to the backend API const response = await fetch('/api/register', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ name, email, type }) }); if (response.ok) { alert('Successfully Registered!'); loadRegistrations(); // Refresh table and counters automatically } } // Function called when page loads to fetch data from MongoDB async function loadRegistrations() { const response = await fetch('/api/registrations'); const data = await response.json(); const tableBody = document.getElementById('tableBody'); tableBody.innerHTML = ''; // Clear previous table content let techCount = 0; let nonTechCount = 0; data.forEach(user => { // Increment counters based on type if (user.type === 'Technical') techCount++; else if (user.type === 'Non-Technical') nonTechCount++; // Add row to table tableBody.innerHTML += ` ${user.name} ${user.email} ${user.type} `; }); // Update Technical and Non-Technical display count document.getElementById('techCount').innerText = techCount; document.getElementById('nonTechCount').innerText = nonTechCount; } // Automatically load database entries when opening or refreshing the page window.onload = loadRegistrations;
